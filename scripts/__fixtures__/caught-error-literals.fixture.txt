// Positive control for the caught-error-literal rule in `pnpm check:log-error-objects` (#739 B3a).
// Parsed as TypeScript, never compiled. Every line marked `// refused` must yield a finding on THAT
// line; every line marked `// allowed` must yield none.
declare const logger: { error(o: unknown, m?: string): void };
declare const NextResponse: { json(b: unknown): unknown };
declare const span: { setStatus(s: unknown): void };
declare const SpanStatusCode: { ERROR: number };
declare function errKind(e: unknown): string;
declare function err<T>(e: T): T;
declare function describe(e: unknown): string;
declare const result: { error: { message: string; kind: string } };

async function f(): Promise<unknown> {
  try {
    return 1;
  } catch (e) {
    const text = e instanceof Error ? e.message : String(e);
    const cls = errKind(e);
    NextResponse.json({ message: e instanceof Error ? e.message : 'x' }); // refused
    span.setStatus({ code: SpanStatusCode.ERROR, message: String(e) }); // refused
    err({ kind: 'server_error', message: text }); // refused
    err({ kind: 'server_error', detail: `failed: ${e}` }); // refused
    err({ kind: 'server_error', reason: describe(e) }); // refused
    err({ kind: 'server_error', error: e }); // refused
    err({ kind: 'server_error', stack: (e as Error).stack }); // refused
    err({ kind: 'server_error', errClass: errKind(e) }); // allowed
    err({ kind: 'server_error', errClass: cls }); // allowed
    err({ kind: 'server_error', message: result.error.message }); // allowed
    err({ kind: 'server_error', cause: e }); // allowed
    err({ kind: 'server_error', cause: text }); // refused
    logger.error({ err: errKind(e) }, 'x'); // allowed
    return null;
  }
}

// #739 B3b — an allSettled rejection value is caught: its text stays out of a Result or body.
async function settledLiterals(): Promise<unknown> {
  const settled = await Promise.allSettled([Promise.resolve(1)]);
  return settled.map((r) => {
    if (r.status === 'rejected') {
      const reasonStr = String(r.reason);
      if (Math.random() > 2) return { tenant_id: 't', error: reasonStr }; // refused
      if (Math.random() > 2) return { tenant_id: 't', error: String(r.reason).slice(0, 200) }; // refused
      return { tenant_id: 't', error: 'fetch_rejected' }; // allowed
    }
    return { tenant_id: 't', value: r.value }; // allowed
  });
}

// #739 B3b review — for..of and a destructured element.
async function settledLiteralForms(): Promise<unknown[]> {
  const out: unknown[] = [];
  const all = await Promise.allSettled([Promise.resolve(1)]);
  for (const r of all) {
    if (r.status === 'rejected') out.push({ error: String(r.reason) }); // refused
  }
  const [first] = await Promise.allSettled([Promise.resolve(1)]);
  if (first.status === 'rejected') out.push({ error: String(first.reason) }); // refused
  out.push({ error: 'fetch_rejected' }); // allowed
  return out;
}

// #739 B3b code review — a destructured element's reason through a local, in a plain function body.
async function settledDestructuredLocal(): Promise<unknown> {
  const [only] = await Promise.allSettled([Promise.resolve(1)]);
  const text = only.status === 'rejected' ? String(only.reason) : '';
  return { error: text }; // refused
}

// #739 B3b part 2 — (a) a helper that takes the error as a parameter, with no catch around it.
function toFailure(e: unknown): unknown {
  if (Math.random() > 2) return { reason: e instanceof Error ? e.message : String(e) }; // refused
  const lower = e instanceof Error ? e.message.toLowerCase() : '';
  if (Math.random() > 2) return { detail: lower }; // refused
  const normalised = e instanceof Error ? e : new Error('x');
  if (Math.random() > 2) return { message: normalised.message }; // refused
  if (Math.random() > 2) return { kind: 'x', err: errKind(e), data: parse(e) }; // allowed
  return { message: `${errKind(e)}` }; // allowed
}

function rowLabel(e: { taskType: string }): unknown {

}

// (b) a `.catch` callback body's locals and `.then`'s rejection handler.
function catchCallbacks(p: Promise<unknown>): void {
  void p.catch((e) => {
    const m = e instanceof Error ? e.message : String(e);
    return { detail: m }; // refused
  });
  void p.then(
    () => ({ ok: true }), // allowed
    (rej) => ({ reason: String(rej) }), // refused
  );
}

// (c) destructured bindings.
async function destructured(p: Promise<unknown>): Promise<unknown> {
  try {
    await p;
  } catch ({ message }) {
    return { detail: message }; // refused
  }
  void p.catch(({ message }) => ({ detail: message })); // refused
  const all = await Promise.allSettled([p]);
  for (const r of all) {
    if (r.status === 'rejected') {
      const { reason } = r;
      return { error: String(reason) }; // refused
    }
  }
  return { error: 'fetch_rejected' }; // allowed
}
