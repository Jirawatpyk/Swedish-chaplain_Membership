// Positive control for `pnpm check:log-error-objects` (spec 140 s2d, #736c).
// Parsed as TypeScript by scripts/lib/log-error-objects-rules.ts, never compiled
// (a .txt so tsc and eslint leave it alone). Every line marked `// refused` must
// yield a finding on THAT line; every line marked `// allowed` must yield none.
// One refused form per line, so a rule that stops matching names its line.
declare const logger: {
  warn(o: unknown, m?: string): void;
  error(o: unknown, m?: string): void;
  child(o: unknown): typeof logger;
};
declare const deps: { logger: typeof logger };
declare const result: { error: { code: string; kind: string; cause: unknown } };
declare function errKind(e: unknown): string;
declare function pgErrorCode(e: unknown): string | undefined;
declare function pgErrorMeta(e: unknown): unknown;
declare function hashId(id: string): string;
declare function classifyThrown(e: unknown): { kind: string; reason?: string };
declare function rootCause(e: unknown): unknown;
declare function describeStorageError(e: unknown): string;
declare const dbError: Error;
declare const sendErr: Error;
declare const cause: unknown;
declare const actorUserId: string;
declare const outcomeKind: string;
declare const errClass: string;
declare const pgCode: string;
declare function describeKind(e: unknown): string;
declare function isBroken(e: unknown): boolean;
declare function approvalErrKind(e: unknown): string;
declare function retryableSubKind(s: string | undefined): string;
declare function isTimeout(e: unknown): boolean;
declare const attempts: number;
declare class GatewayThrowable extends Error {
  readonly kind: string;
  readonly subKind?: string;
  readonly reason: string;
}
declare class UnlistedError extends Error {
  readonly detailText: string;
}

export async function refusedForms(): Promise<void> {
  try {
    await Promise.resolve();
  } catch (e) {
    logger.error({ err: e }, 'x'); // refused
    logger.error({ err: e.message }, 'x'); // refused
    logger.error({ stack: e.stack }, 'x'); // refused
    logger.error({ err: String(e) }, 'x'); // refused
    logger.error({ err: JSON.stringify(e) }, 'x'); // refused
    logger.error({ err: `failed: ${e.message}` }, 'x'); // refused
    logger.error({ err: e instanceof Error ? e.message : 'unknown' }, 'x'); // refused
    logger.error({ err: e instanceof Error ? errKind(e) : String(e) }, 'x'); // refused
    logger.error({ ...e }, 'x'); // refused
    logger.error(e, 'x'); // refused
    logger.error({ nested: { err: e } }, 'x'); // refused
    logger.error({ err: describeStorageError(e) }, 'x'); // refused
    logger.error({ err: classifyThrown(e) }, 'x'); // refused
    logger.error({ why: classifyThrown(e).reason }, 'x'); // refused
    logger.error({ err: errKind(e) }, `failed: ${e.message}`); // refused
    deps.logger.warn({ err: e }, 'x'); // refused
    // Review round 1 (security M1, code-review #1, security L2, L3) — the stricter rule.
    logger.error({ err: e['message'] }, 'x'); // refused
    logger.error({ err: result.error['cause'] }, 'x'); // refused
    logger.error({ err: e.toString() }, 'x'); // refused
    logger.error({ err: e.message.slice(0, 40) }, 'x'); // refused
    logger.error({ err: (e as Error).message.trim() }, 'x'); // refused
    logger.error({ err: JSON.stringify({ e }) }, 'x'); // refused
    logger.error({ err: JSON.stringify({ failure: e }) }, 'x'); // refused
    logger.error({ err: String([e]) }, 'x'); // refused
    logger.error({ params: e.params }, 'x'); // refused
    logger.error({ detail: result.error.cause.detail }, 'x'); // refused
    logger.error({ err: describeKind(e) }, 'x'); // refused
    logger.error({ broken: isBroken(e) }, 'x'); // refused
    logger.child({ err: e }); // refused
    deps.logger.child({ cause: e.message }); // refused
    // Review round 2 (security #4, code-review #8) — one hop through a local.
    const { message } = e;
    logger.warn({ message }); // refused
    const m = e.message;
    logger.warn({ m }); // refused
    const { detail: d, ...rest } = e;
    logger.warn({ d }); // refused
    logger.warn({ rest }); // refused
    const alias = e;
    logger.warn({ failure: alias }); // refused
    const reasonText = e instanceof GatewayThrowable ? e.reason : undefined;
    logger.warn({ reasonText }); // refused
    const wrongField: GatewayThrowable['subKind'] = e.reason;
    logger.warn({ wrongField }); // refused
    const unlisted: UnlistedError['detailText'] = e.detailText;
    logger.warn({ unlisted }); // refused
  }
  {
    const { errClass } = result.error;
    logger.error({ err: errClass }); // refused
    const causeClass = result.error.cause;
    logger.error({ err: causeClass }); // refused
  }
  try {
    await Promise.resolve();
  } catch (errClass) {
    logger.error({ errClass }, 'x'); // refused
  }
  try {
    await Promise.resolve();
  } catch (thrown) {
    logger.error({ err: thrown }, 'x'); // refused
  }
  await Promise.resolve().catch((boom) => logger.error({ err: boom }, 'x')); // refused
  const err = new Error('x');
  const error = new Error('x');
  const ex = new Error('x');
  const e = new Error('x');
  logger.error({ err }, 'x'); // refused
  logger.error({ error }, 'x'); // refused
  logger.error({ cause }, 'x'); // refused
  logger.error({ e }, 'x'); // refused
  logger.error({ failure: ex }, 'x'); // refused
  logger.error({ failure: dbError }, 'x'); // refused
  logger.error({ failure: sendErr }, 'x'); // refused
  logger.error({ err: result.error }, 'x'); // refused
  logger.error({ err: result.error.cause }, 'x'); // refused
  logger.error({ err: rootCause(result.error) }, 'x'); // refused
  logger.error({ err: error.message ?? 'none' }, 'x'); // refused
}

export async function allowedForms(): Promise<void> {
  try {
    await Promise.resolve();
  } catch (e) {
    logger.error({ err: errKind(e), code: pgErrorCode(e) }, 'x'); // allowed
    logger.error({ err: errKind(rootCause(e)), ...{ meta: pgErrorMeta(e) } }, 'x'); // allowed
    logger.error({ kind: classifyThrown(e).kind }, 'x'); // allowed
    logger.error({ name: e.name, kind: e.kind, code: e.code }, 'x'); // allowed
    logger.error({ err: approvalErrKind(e), timedOut: isTimeout(e) }, 'x'); // allowed
    logger.error({ sub: retryableSubKind(classifyThrown(e).subKind), finite: Number.isFinite(attempts) }, 'x'); // allowed
    logger.child({ err: errKind(e), code: pgErrorCode(e) }); // allowed
    // Review round 2 — a token field read into a local stays a token; a
    // reviewed class field is allowed only when annotated with its class.
    const { code: c, kind: k, name: n } = e;
    logger.error({ c, k, n }); // allowed
    const subKind: GatewayThrowable['subKind'] | undefined = e instanceof GatewayThrowable ? e.subKind : undefined;
    logger.error({ subKind }); // allowed
  }
  logger.error({ err: errKind(rootCause(result.error)), repoCode: result.error.code }, 'x'); // allowed
  logger.error({ kind: result.error.kind }, 'x'); // allowed
  logger.error({ actor: hashId(actorUserId) }, 'x'); // allowed
  logger.error({ outcomeKind, errClass, pgCode }, 'x'); // allowed
  logger.error({ err: errClass, step: 'insert_row' }, 'x'); // allowed
}
